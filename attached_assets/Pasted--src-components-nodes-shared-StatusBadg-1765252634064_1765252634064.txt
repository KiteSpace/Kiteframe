src/components/nodes/shared/StatusBadge.tsx

import React from 'react';
import './StatusBadge.css';

export function StatusBadge({
  status,
  onClick,
  disabled,
}: {
  status?: 'todo' | 'inprogress' | 'done';
  onClick?: () => void;
  disabled?: boolean;
}) {
  const label =
    status === 'inprogress'
      ? 'In Progress'
      : status === 'done'
      ? 'Done'
      : 'To Do';

  return (
    <div
      className={`kf-status-badge kf-${status ?? 'todo'} ${
        disabled ? 'kf-disabled' : ''
      }`}
      onClick={disabled ? undefined : onClick}
    >
      <span className="dot" /> {label}
    </div>
  );
}

============================================================
Add Status Badge to Node Footer (Example)
============================================================

Inside ANY node renderer, find the footer section:

<footer className="kf-node-footer">
  {isStatusEnabled && (
    <StatusBadge
      status={node.data?.status}
      onClick={() => onStatusClick(node.id)}
      disabled={readOnly}
    />
  )}
</footer>


Works for all nodes: Basic, Form, Code, Table, Compound.

============================================================
Cycle Function
============================================================

src/utils/status.ts

export function cycleStatus(
  status?: 'todo' | 'inprogress' | 'done'
): 'todo' | 'inprogress' | 'done' {
  if (!status || status === 'todo') return 'inprogress';
  if (status === 'inprogress') return 'done';
  return 'todo';
}

============================================================
Canvas Integration
============================================================

Inside Editor/Canvas controller:

function handleNodeStatusClick(id: string) {
  setNodes(prev =>
    prev.map(n =>
      n.id === id
        ? { ...n, data: { ...n.data, status: cycleStatus(n.data?.status) } }
        : n
    )
  );
}

============================================================
Workflow Controls
============================================================

When enabling tracking:

workflow.statusTrackingEnabled = true;

workflow.nodeIds.forEach(id => {
  const node = nodes.find(n => n.id === id);
  if (node && !node.data?.status) {
    node.data.status = 'todo';
  }
});


When disabling:

workflow.statusTrackingEnabled = false;
// badges simply do not render
