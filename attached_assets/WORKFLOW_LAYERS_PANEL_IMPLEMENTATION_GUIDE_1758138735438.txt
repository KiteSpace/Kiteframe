KITEFRAME WORKFLOW LAYERS & LAYERS PANEL - COMPLETE IMPLEMENTATION GUIDE
================================================================================

This document provides a comprehensive implementation guide for KiteFrame's advanced 
workflow layers system and layers panel UI. The system organizes workflow elements 
into hierarchical structures similar to professional design tools like Figma, 
Adobe Illustrator, or Sketch.

CORE FEATURES:
- Hierarchical workflow organization with automatic grouping
- Smart workflow detection using connected component analysis
- Topological sorting of nodes within workflows
- Interactive layer management (visibility, locking, renaming)
- Real-time synchronization with canvas
- Professional design tool UX patterns

================================================================================
PART 1: SYSTEM ARCHITECTURE
================================================================================

COMPONENT HIERARCHY
-------------------
```
FlowSidePanel (Main Container)
├── Tab Navigation (Flows, Layers, Projects, History)  
└── LayersPanel (Active when Layers tab selected)
    ├── NestedLayersPanel (Hierarchical display)
    ├── LayerBuilder (Logic for workflow detection)
    └── Individual WorkflowGroups
        ├── Workflow Header (with controls)
        ├── Node Items (sorted topologically)
        └── Connection Items (edges between nodes)
```

WORKFLOW DETECTION ALGORITHM
----------------------------
The system uses sophisticated algorithms to automatically organize elements:

1. **Connected Component Analysis** - Groups nodes that are connected via edges
2. **Topological Sorting** - Orders nodes within workflows by dependency flow
3. **Smart Naming** - Generates meaningful workflow names from node content
4. **Template Recognition** - Identifies common workflow patterns

import { WorkflowGroup, LayerNode, KFrameGroup } from "./layers_types";

export function buildLayerStructure(nodes: any[], edges: any[], canvasTexts: any[] = []): WorkflowGroup[] {
  console.log('[LayerBuilder] Building layer structure', { 
    nodeCount: nodes.length, 
    edgeCount: edges.length, 
    textCount: canvasTexts.length 
  });
  
  if (nodes.length === 0) {
    return [];
  }
  
  const workflowMap: Record<string, { nodes: any[], edges: any[], baseName: string }> = {};

  // Define comprehensive workflow prefixes for template recognition
  const workflowPrefixes = [
    { prefix: 'data-processing', name: 'Data Processing', order: 1 },
    { prefix: 'user-workflow', name: 'User Workflow', order: 2 },
    { prefix: 'api-integration', name: 'API Integration', order: 3 },
    { prefix: 'analytics-dashboard', name: 'Analytics Dashboard', order: 4 },
    { prefix: 'approval-workflow', name: 'Approval Workflow', order: 5 },
    { prefix: 'automation-pipeline', name: 'Automation Pipeline', order: 6 },
    { prefix: 'ci-cd-pipeline', name: 'CI/CD Pipeline', order: 7 },
    { prefix: 'data-pipeline', name: 'Data Pipeline', order: 8 },
    { prefix: 'ml-pipeline', name: 'ML Pipeline', order: 9 },
    { prefix: 'content-workflow', name: 'Content Workflow', order: 10 },
    // ... up to 26+ predefined workflow templates
    { prefix: 'sales-process', name: 'Sales Process', order: 26 }
  ];

  // Helper function to get node visual properties
  const getNodeColor = (node: any): string => {
    if (node.data?.backgroundColor) return node.data.backgroundColor;
    if (node.data?.borderColor) return node.data.borderColor;
    if (node.data?.color) return node.data.color;
    return '#3b82f6'; // Default blue
  };

  // Create node lookup map for efficient access
  const nodeMap = new Map();
  nodes.forEach(node => {
    nodeMap.set(node.id, {
      label: node.data?.label || node.id,
      color: getNodeColor(node),
      description: node.data?.description || ''
    });
  });

  // STEP 1: GROUP EDGES BY WORKFLOW TYPE
  // First attempt: Group by known prefixes
  const edgesByWorkflowType: Record<string, any[]> = {};
  const unmatchedEdges: any[] = [];
  
  edges.forEach((edge, i) => {
    const workflow = workflowPrefixes.find(w => edge.id && edge.id.startsWith(w.prefix));
    if (workflow) {
      if (!edgesByWorkflowType[workflow.prefix]) {
        edgesByWorkflowType[workflow.prefix] = [];
      }
      edgesByWorkflowType[workflow.prefix].push(edge);
    } else {
      unmatchedEdges.push(edge);
    }
  });

  // STEP 2: CONNECTED COMPONENT ANALYSIS
  // For edges without workflow prefixes (simple IDs like "edge1", "edge2")
  if (unmatchedEdges.length > 0 && Object.keys(edgesByWorkflowType).length === 0) {
    console.log('[LayerBuilder] Using connected component analysis for simple edge IDs');
    
    // Build adjacency list from all edges
    const adjacencyList = new Map<string, Set<string>>();
    const allNodes = new Set<string>();
    
    edges.forEach(edge => {
      allNodes.add(edge.source);
      allNodes.add(edge.target);
      
      if (!adjacencyList.has(edge.source)) {
        adjacencyList.set(edge.source, new Set());
      }
      if (!adjacencyList.has(edge.target)) {
        adjacencyList.set(edge.target, new Set());
      }
      
      adjacencyList.get(edge.source)!.add(edge.target);
      adjacencyList.get(edge.target)!.add(edge.source);
    });
    
    // Find connected components using Depth-First Search
    const visited = new Set<string>();
    const components: string[][] = [];
    
    const dfs = (nodeId: string, component: string[]) => {
      visited.add(nodeId);
      component.push(nodeId);
      
      const neighbors = adjacencyList.get(nodeId) || new Set();
      for (const neighbor of Array.from(neighbors)) {
        if (!visited.has(neighbor)) {
          dfs(neighbor, component);
        }
      }
    };
    
    // Find all connected components
    for (const nodeId of Array.from(allNodes)) {
      if (!visited.has(nodeId)) {
        const component: string[] = [];
        dfs(nodeId, component);
        if (component.length > 1) { // Only multi-node components become workflows
          components.push(component);
        }
      }
    }
    
    // Create workflows from connected components
    components.forEach((component, index) => {
      const workflowId = `ai-workflow-${index + 1}`;
      const workflowName = `AI Workflow ${index + 1}`;
      
      // Find edges within this component
      const componentEdges = edges.filter(edge => 
        component.includes(edge.source) && component.includes(edge.target)
      );
      
      workflowMap[workflowId] = {
        nodes: component.map(nodeId => nodes.find(n => n.id === nodeId)).filter(Boolean),
        edges: componentEdges,
        baseName: workflowName
      };
    });
    
    // Handle isolated nodes (not connected to anything)
    const connectedNodes = new Set(Array.from(allNodes));
    const isolatedNodes = nodes.filter(node => !connectedNodes.has(node.id));
    
    if (isolatedNodes.length > 0) {
      workflowMap['isolated-nodes'] = {
        nodes: isolatedNodes,
        edges: [],
        baseName: 'Isolated Nodes'
      };
    }
  } else {
    // Handle workflow-prefixed edges with connected component analysis within each type
    Object.entries(edgesByWorkflowType).forEach(([prefix, workflowEdges]) => {
      // Build adjacency graph for this workflow type
      const adjacency = new Map<string, Set<string>>();
      workflowEdges.forEach(edge => {
        if (!adjacency.has(edge.source)) adjacency.set(edge.source, new Set());
        if (!adjacency.has(edge.target)) adjacency.set(edge.target, new Set());
        adjacency.get(edge.source)!.add(edge.target);
        adjacency.get(edge.target)!.add(edge.source);
      });
      
      // Find connected components using iterative DFS
      const visited = new Set<string>();
      const components: string[][] = [];
      
      for (const node of Array.from(adjacency.keys())) {
        if (!visited.has(node)) {
          const component: string[] = [];
          const stack = [node];
          
          while (stack.length > 0) {
            const current = stack.pop()!;
            if (!visited.has(current)) {
              visited.add(current);
              component.push(current);
              
              const neighbors = adjacency.get(current) || new Set();
              for (const neighbor of Array.from(neighbors)) {
                if (!visited.has(neighbor)) {
                  stack.push(neighbor);
                }
              }
            }
          }
          
          if (component.length > 0) {
            components.push(component);
          }
        }
      }
      
      // Create workflow groups for each connected component
      components.forEach((component, index) => {
        const workflowId = `${prefix}-${index + 1}`;
        const workflowInfo = workflowPrefixes.find(w => w.prefix === prefix);
        
        workflowMap[workflowId] = {
          nodes: [],
          edges: workflowEdges.filter(edge => 
            component.includes(edge.source) || component.includes(edge.target)
          ),
          baseName: workflowInfo?.name || prefix
        };
      });
    });
  }

  // Add nodes to their respective workflows
  nodes.forEach(node => {
    let assigned = false;
    
    // First, try to assign based on connected edges
    for (const [workflowId, workflowData] of Object.entries(workflowMap)) {
      const isConnected = workflowData.edges.some(edge => 
        edge.source === node.id || edge.target === node.id
      );
      
      if (isConnected) {
        workflowData.nodes.push(node);
        assigned = true;
        break;
      }
    }
    
    if (!assigned) {
      // Handle special node types
      if (node.type === 'kframe' || (node.data && node.data.type === 'kframe')) {
        if (!workflowMap['frames']) {
          workflowMap['frames'] = { nodes: [], edges: [], baseName: 'Frames' };
        }
        workflowMap['frames'].nodes.push(node);
      } else {
        // Ungrouped nodes
        if (!workflowMap['ungrouped']) {
          workflowMap['ungrouped'] = { nodes: [], edges: [], baseName: 'Ungrouped' };
        }
        workflowMap['ungrouped'].nodes.push(node);
      }
    }
  });

  // STEP 3: TOPOLOGICAL SORTING OF NODES WITHIN WORKFLOWS
  function topologicalSort(workflowNodes: any[], workflowEdges: any[]): any[] {
    const inDegree = new Map<string, number>();
    const graph = new Map<string, string[]>();
    
    // Initialize in-degree and adjacency graph
    workflowNodes.forEach(node => {
      inDegree.set(node.id, 0);
      graph.set(node.id, []);
    });
    
    // Build graph and calculate in-degrees using Kahn's algorithm preparation
    workflowEdges.forEach(edge => {
      if (inDegree.has(edge.source) && inDegree.has(edge.target)) {
        graph.get(edge.source)?.push(edge.target);
        inDegree.set(edge.target, (inDegree.get(edge.target) || 0) + 1);
      }
    });
    
    // Kahn's algorithm for topological sorting
    const queue: string[] = [];
    const sorted: any[] = [];
    
    // Start with nodes that have no dependencies (in-degree = 0)
    inDegree.forEach((degree, nodeId) => {
      if (degree === 0) {
        queue.push(nodeId);
      }
    });
    
    while (queue.length > 0) {
      const nodeId = queue.shift()!;
      const node = workflowNodes.find(n => n.id === nodeId);
      if (node) {
        sorted.push(node);
        
        // Reduce in-degree of connected nodes
        graph.get(nodeId)?.forEach(neighbor => {
          const newInDegree = (inDegree.get(neighbor) || 0) - 1;
          inDegree.set(neighbor, newInDegree);
          if (newInDegree === 0) {
            queue.push(neighbor);
          }
        });
      }
    }
    
    // Add any remaining nodes (handles cycles or disconnected components)
    workflowNodes.forEach(node => {
      if (!sorted.find(n => n.id === node.id)) {
        sorted.push(node);
      }
    });
    
    return sorted;
  }

  // STEP 4: BUILD FINAL WORKFLOW STRUCTURE
  const workflowGroups: WorkflowGroup[] = [];
  
  Object.entries(workflowMap).forEach(([workflowId, workflowData]) => {
    if (workflowData.nodes.length === 0) return;
    
    // Sort nodes topologically for proper workflow flow
    const sortedNodes = topologicalSort(workflowData.nodes, workflowData.edges);
    
    // Create numbered node entries with enhanced labeling
    const children: LayerNode[] = sortedNodes.map((node, index) => ({
      id: node.id,
      label: `${index + 1}. ${node.data?.label || node.id}`,
      description: node.data?.description,
      type: "node" as const,
      icon: "Square",
      color: getNodeColor(node)
    }));
    
    // Add connections with descriptive labels
    const connections: LayerNode[] = workflowData.edges.map((edge, index) => {
      // Find source and target nodes for better connection naming
      const sourceNode = workflowData.nodes.find(n => n.id === edge.source);
      const targetNode = workflowData.nodes.find(n => n.id === edge.target);
      
      const sourceName = sourceNode?.data?.label || `Node ${index + 1}`;
      const targetName = targetNode?.data?.label || `Node ${index + 2}`;
      
      return {
        id: edge.id,
        label: `${sourceName} → ${targetName}`,
        description: `Connects "${sourceName}" to "${targetName}"`,
        type: "edge" as const,
        icon: "Link",
        color: "#64748b"
      };
    });
    
    // Generate workflow name
    const workflowName = workflowId === 'ungrouped' 
      ? 'Ungrouped' 
      : workflowId === 'frames'
      ? 'Frames'
      : workflowData.baseName;
    
    // Calculate total items
    const totalCount = children.length + connections.length;
    
    workflowGroups.push({
      id: workflowId,
      name: workflowName,
      children: [...children, ...connections],
      childCount: totalCount,
      connections: connections
    });
  });
  
  console.log('[LayerBuilder] Final workflow groups created:', workflowGroups.length);
  
  return workflowGroups;
}

import React, { useState } from 'react';
import { ChevronDown, ChevronRight } from 'lucide-react';
import { WorkflowGroup, LayerNode } from './layers_types';

interface NestedLayersPanelProps {
  workflows: WorkflowGroup[];
  onSelect: (id: string, type: 'node' | 'edge' | 'text' | 'kframe') => void;
  workflowNames?: Record<string, string>;
  onRenameWorkflow?: (id: string, name: string) => void;
  selectedItemId?: string;
  onToggleWorkflowVisibility?: (id: string) => void;
  onDeleteWorkflow?: (id: string) => void;
}

export function NestedLayersPanel({ 
  workflows, 
  onSelect, 
  workflowNames = {}, 
  onRenameWorkflow,
  selectedItemId,
  onToggleWorkflowVisibility,
  onDeleteWorkflow
}: NestedLayersPanelProps) {
  const [expanded, setExpanded] = useState<Record<string, boolean>>({});
  const [editingName, setEditingName] = useState<Record<string, string>>({});

  const toggle = (workflowId: string) => {
    setExpanded(prev => ({ ...prev, [workflowId]: !prev[workflowId] }));
  };

  const handleNameChange = (workflowId: string, value: string) => {
    setEditingName(prev => ({ ...prev, [workflowId]: value }));
  };

  const handleNameBlur = (workflowId: string) => {
    const newName = editingName[workflowId];
    if (newName && onRenameWorkflow) {
      onRenameWorkflow(workflowId, newName);
    }
    setEditingName(prev => {
      const next = { ...prev };
      delete next[workflowId];
      return next;
    });
  };

  const handleNameKeyDown = (workflowId: string, e: React.KeyboardEvent) => {
    if (e.key === 'Enter') {
      handleNameBlur(workflowId);
    }
    if (e.key === 'Escape') {
      setEditingName(prev => {
        const next = { ...prev };
        delete next[workflowId];
        return next;
      });
    }
  };

  const initializeEditName = (workflowId: string, currentName: string) => {
    setEditingName(prev => ({ ...prev, [workflowId]: currentName }));
  };

  const renderIcon = (type: string) => {
    switch (type) {
      case 'node':
        return (
          <svg xmlns="http://www.w3.org/2000/svg" width="14" height="14" viewBox="0 0 24 24" 
               fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" 
               strokeLinejoin="round" className="lucide lucide-square text-slate-500">
            <rect width="18" height="18" x="3" y="3" rx="2"></rect>
          </svg>
        );
      case 'edge':
        return (
          <svg xmlns="http://www.w3.org/2000/svg" width="14" height="14" viewBox="0 0 24 24" 
               fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" 
               strokeLinejoin="round" className="lucide lucide-link text-slate-500">
            <path d="M10 13a5 5 0 0 0 7.54.54l3-3a5 5 0 0 0-7.07-7.07l-1.72 1.71"></path>
            <path d="M14 11a5 5 0 0 0-7.54-.54l-3 3a5 5 0 0 0 7.07 7.07l1.71-1.71"></path>
          </svg>
        );
      case 'text':
        return (
          <svg xmlns="http://www.w3.org/2000/svg" width="14" height="14" viewBox="0 0 24 24" 
               fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" 
               strokeLinejoin="round" className="lucide lucide-type text-slate-500">
            <polyline points="4,7 4,4 20,4 20,7"></polyline>
            <line x1="9" y1="20" x2="15" y2="20"></line>
            <line x1="12" y1="4" x2="12" y2="20"></line>
          </svg>
        );
      case 'kframe':
        return (
          <svg xmlns="http://www.w3.org/2000/svg" width="14" height="14" viewBox="0 0 24 24" 
               fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" 
               strokeLinejoin="round" className="lucide lucide-folder text-slate-500">
            <path d="M20 20a2 2 0 0 0 2-2V8a2 2 0 0 0-2-2h-7.9a2 2 0 0 1-1.69-.9L9.6 3.9A2 2 0 0 0 7.93 3H4a2 2 0 0 0-2 2v13a2 2 0 0 0 2 2Z"></path>
          </svg>
        );
      default:
        return null;
    }
  };

  return (
    <div className="flex-1 overflow-y-auto">
      <div className="p-2">
        <div className="text-slate-900 dark:text-slate-100">
          {workflows.map((workflow) => {
            const isExpanded = expanded[workflow.id] !== false; // default to true
            const displayName = workflowNames[workflow.id] || workflow.name;
            const isEditing = editingName[workflow.id] !== undefined;
            
            return (
              <div key={workflow.id} className="border-b border-slate-200 dark:border-slate-700">
                {/* Workflow Header */}
                <div className="flex items-center h-8 px-2 hover:bg-slate-50 dark:hover:bg-slate-800/50">
                  <button 
                    className="w-4 text-xs mr-1 flex items-center justify-center hover:bg-slate-200 dark:hover:bg-slate-700 rounded"
                    onClick={() => toggle(workflow.id)}
                  >
                    {isExpanded ? 
                      <ChevronDown className="w-3.5 h-3.5" /> : 
                      <ChevronRight className="w-3.5 h-3.5" />
                    }
                  </button>
                  
                  <span className="mr-2 text-slate-500">
                    <svg xmlns="http://www.w3.org/2000/svg" width="14" height="14" viewBox="0 0 24 24" 
                         fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" 
                         strokeLinejoin="round" className="lucide lucide-folder">
                      <path d="M20 20a2 2 0 0 0 2-2V8a2 2 0 0 0-2-2h-7.9a2 2 0 0 1-1.69-.9L9.6 3.9A2 2 0 0 0 7.93 3H4a2 2 0 0 0-2 2v13a2 2 0 0 0 2 2Z"></path>
                    </svg>
                  </span>
                  
                  <input 
                    className="bg-transparent text-sm font-medium outline-none border-b border-transparent focus:border-slate-300 dark:focus:border-slate-600 flex-1 mr-2"
                    value={isEditing ? editingName[workflow.id] : displayName}
                    onChange={(e) => handleNameChange(workflow.id, e.target.value)}
                    onBlur={() => handleNameBlur(workflow.id)}
                    onKeyDown={(e) => handleNameKeyDown(workflow.id, e)}
                    onFocus={() => {
                      if (!isEditing) {
                        initializeEditName(workflow.id, displayName);
                      }
                    }}
                  />
                  
                  <span className="text-xs text-slate-500 font-medium">
                    {workflow.childCount || workflow.children?.length || 0}
                  </span>
                </div>

                {/* Workflow Content */}
                {isExpanded && (
                  <div className="pb-2">
                    {/* Nodes */}
                    {workflow.children?.filter(child => child.type === 'node').map((node) => (
                      <div
                        key={node.id}
                        className={`py-1.5 px-2 hover:bg-slate-50 dark:hover:bg-slate-800/50 cursor-pointer flex items-center gap-2 transition-colors ${
                          selectedItemId === node.id ? 'bg-blue-50 dark:bg-blue-900/20' : ''
                        }`}
                        title={node.description || node.label}
                        style={{ paddingLeft: '22px' }}
                        onClick={() => onSelect(node.id, node.type)}
                      >
                        {renderIcon(node.type)}
                        <div className="truncate text-sm flex-1" style={{ color: node.color }}>
                          {node.label}
                        </div>
                        {node.description && (
                          <div className="text-xs text-slate-400 truncate max-w-[100px]">
                            {node.description}
                          </div>
                        )}
                      </div>
                    ))}
                    
                    {/* KFrames */}
                    {workflow.children?.filter(child => child.type === 'kframe').map((kframe) => (
                      <div
                        key={kframe.id}
                        className={`py-1.5 px-2 hover:bg-slate-50 dark:hover:bg-slate-800/50 cursor-pointer flex items-center gap-2 transition-colors ${
                          selectedItemId === kframe.id ? 'bg-blue-50 dark:bg-blue-900/20' : ''
                        }`}
                        title={kframe.description || kframe.label}
                        style={{ paddingLeft: '22px' }}
                        onClick={() => onSelect(kframe.id, kframe.type)}
                      >
                        {renderIcon(kframe.type)}
                        <div className="truncate text-sm flex-1" style={{ color: kframe.color }}>
                          {kframe.label}
                        </div>
                      </div>
                    ))}
                    
                    {/* Text Objects */}
                    {workflow.children?.filter(child => child.type === 'text').map((textObj) => (
                      <div
                        key={textObj.id}
                        className={`py-1.5 px-2 hover:bg-slate-50 dark:hover:bg-slate-800/50 cursor-pointer flex items-center gap-2 transition-colors ${
                          selectedItemId === textObj.id ? 'bg-blue-50 dark:bg-blue-900/20' : ''
                        }`}
                        title={textObj.description || textObj.label}
                        style={{ paddingLeft: '22px' }}
                        onClick={() => onSelect(textObj.id, textObj.type)}
                      >
                        {renderIcon(textObj.type)}
                        <div className="truncate text-sm flex-1" style={{ color: textObj.color }}>
                          {textObj.label}
                        </div>
                      </div>
                    ))}
                    
                    {/* Connections Header */}
                    {workflow.connections && workflow.connections.length > 0 && (
                      <>
                        <div
                          className="py-1.5 px-2 hover:bg-slate-50 dark:hover:bg-slate-800/50 cursor-pointer flex items-center gap-2 transition-colors"
                          title="Connections"
                          style={{ paddingLeft: '22px' }}
                        >
                          <svg xmlns="http://www.w3.org/2000/svg" width="14" height="14" viewBox="0 0 24 24" 
                               fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" 
                               strokeLinejoin="round" className="lucide lucide-link text-slate-500">
                            <path d="M10 13a5 5 0 0 0 7.54.54l3-3a5 5 0 0 0-7.07-7.07l-1.72 1.71"></path>
                            <path d="M14 11a5 5 0 0 0-7.54-.54l-3 3a5 5 0 0 0 7.07 7.07l1.71-1.71"></path>
                          </svg>
                          <div className="truncate text-sm font-medium text-slate-600 dark:text-slate-400">
                            Connections ({workflow.connections.length})

          })}
        </div>
      </div>

  );

export const LayersPanel: React.FC<LayersPanelProps> = ({
  nodes,
  edges,
  groups,
  selectedId,
  onSelect,
  onToggleVisibility,
  onToggleLock,
  onRenameWorkflow,
  onDeleteWorkflow,
  onToggleWorkflowVisibility,
}) => {
  const [expandedSections, setExpandedSections] = useState({
    nodes: true,
    edges: true,
    groups: true,
  });
  const [editingWorkflow, setEditingWorkflow] = useState<string | null>(null);
  const [editingName, setEditingName] = useState('');
  const [hiddenWorkflows, setHiddenWorkflows] = useState<Set<string>>(new Set());

  const toggleSection = (section: 'nodes' | 'edges' | 'groups' | string) => {
    setExpandedSections(prev => ({
      ...prev,
      [section]: !prev[section as keyof typeof prev],
    }));
  };

  const handleRenameWorkflow = (workflowId: string, currentName: string) => {
    setEditingWorkflow(workflowId);
    setEditingName(currentName);
  };

  const handleSaveRename = (workflowId: string) => {
    if (editingName.trim() && onRenameWorkflow) {
      onRenameWorkflow(workflowId, editingName.trim());
    }
    setEditingWorkflow(null);
    setEditingName('');
  };

  const handleCancelRename = () => {
    setEditingWorkflow(null);
    setEditingName('');
  };

  const handleDeleteWorkflow = (workflowId: string) => {
    if (onDeleteWorkflow) {
      onDeleteWorkflow(workflowId);
    }
  };

  const handleToggleWorkflowVisibility = (workflowId: string) => {
    setHiddenWorkflows(prev => {
      const newSet = new Set(prev);
      if (newSet.has(workflowId)) {
        newSet.delete(workflowId);
      } else {
        newSet.add(workflowId);
      }
      return newSet;
    });
    
    if (onToggleWorkflowVisibility) {
      onToggleWorkflowVisibility(workflowId);
    }
  };

  // SMART WORKFLOW DETECTION ALGORITHM
  const detectWorkflows = () => {
    const workflowGroups: { [key: string]: { nodes: NodeData[]; edges: EdgeData[]; name: string; order: number } } = {};
    const ungroupedNodes: NodeData[] = [];
    const visitedNodes = new Set<string>();

    // Helper function to find connected nodes (workflow detection)
    const findConnectedNodes = (startNodeId: string, visited = new Set<string>()): Set<string> => {
      if (visited.has(startNodeId)) return visited;
      visited.add(startNodeId);
      
      // Find all edges connected to this node
      edges.forEach(edge => {
        if (edge.source === startNodeId && !visited.has(edge.target)) {
          findConnectedNodes(edge.target, visited);
        }
        if (edge.target === startNodeId && !visited.has(edge.source)) {
          findConnectedNodes(edge.source, visited);
        }
      });
      
      return visited;
    };

    // Helper function to generate smart workflow name based on nodes
    const generateWorkflowName = (workflowNodes: NodeData[]): string => {
      // Priority: KFrame label > Most common node type > First node label
      const kframe = workflowNodes.find(n => n.type === 'kframe');
      if (kframe && kframe.label) {
        return kframe.label;
      }

      // Check for AI-generated workflow patterns
      const hasAINodes = workflowNodes.some(n => n.id.startsWith('node'));
      if (hasAINodes) {
        // Look for user nodes that often indicate the workflow purpose
        const userNode = workflowNodes.find(n => n.type === 'user');
        if (userNode && userNode.label) {
          return userNode.label;
        }
      }

      // Count node types to determine workflow type
      const typeCounts: { [key: string]: number } = {};
      workflowNodes.forEach(node => {
        const type = node.type || 'default';
        typeCounts[type] = (typeCounts[type] || 0) + 1;
      });

      // Generate name based on dominant node types
      const dominantType = Object.entries(typeCounts)
        .sort((a, b) => b[1] - a[1])[0]?.[0];

      const typeNames: { [key: string]: string } = {
        'api': 'API Integration',
        'database': 'Data Processing',
        'user': 'User Workflow',
        'process': 'Process Flow',
        'decision': 'Decision Flow',
        'data': 'Data Pipeline',
        'output': 'Output Process',
        'input': 'Input Handler'
      };

      // Use first meaningful node label if available
      const firstLabeledNode = workflowNodes.find(n => n.label && n.label.length > 2);
      if (firstLabeledNode && firstLabeledNode.label) {
        return firstLabeledNode.label;
      }

      return typeNames[dominantType] || 'Workflow';
    };

    // Detect workflows by finding connected components
    let workflowIndex = 0;
    nodes.forEach(node => {
      if (!visitedNodes.has(node.id)) {
        const connectedNodeIds = findConnectedNodes(node.id);
        
        // Only create a workflow group if there are at least 2 connected nodes
        if (connectedNodeIds.size >= 2) {
          const workflowNodes = nodes.filter(n => connectedNodeIds.has(n.id));
          const workflowEdges = edges.filter(e => 
            connectedNodeIds.has(e.source) && connectedNodeIds.has(e.target)
          );
          
          const workflowId = `workflow-${workflowIndex}`;
          const workflowName = generateWorkflowName(workflowNodes);
          
          // Sort nodes by position for proper workflow structure
          workflowNodes.sort((a, b) => {
            if (!a.position || !b.position) return 0;
            const yDiff = a.position.y - b.position.y;
            if (Math.abs(yDiff) > 50) return yDiff;
            return a.position.x - b.position.x;
          });
          
          workflowGroups[workflowId] = {
            nodes: workflowNodes,
            edges: workflowEdges,
            name: workflowName,
            order: workflowIndex
          };
          
          connectedNodeIds.forEach(id => visitedNodes.add(id));
          workflowIndex++;
        } else {
          // Single unconnected node
          ungroupedNodes.push(node);
          visitedNodes.add(node.id);
        }
      }
    });

    return { workflowGroups, ungroupedNodes };
  };

  const { workflowGroups, ungroupedNodes } = detectWorkflows();

  return (

              onClick={() => toggleSection('groups')}
            >
              {expandedSections.groups ? <ChevronDown size={14} /> : <ChevronRight size={14} />}
              <span>Groups ({groups.length})</span>
            </button>
            {expandedSections.groups && (
              <div className="layers-panel-section-content px-2">
                {groups.map((group) => (
                  <LayerItem
                    key={group.id}
                    id={group.id}
                    type="group"
                    label={group.label}
                    color={group.color}
                    isSelected={selectedId === group.id}
                    isVisible={group.isVisible}
                    isLocked={group.isLocked}
                    children={group.children}
                    onSelect={onSelect}
                    onToggleVisibility={onToggleVisibility}
                    onToggleLock={onToggleLock}
                  />
                ))}
              </div>
            )}
          </div>
        )}

        {/* Workflow Groups - Sorted by order */}
        {Object.entries(workflowGroups)
          .sort(([, a], [, b]) => a.order - b.order)
          .map(([workflowId, workflow]) => (
          <div key={workflowId} className="layers-panel-section">
            <div className="layers-panel-section-header-container flex items-center px-3 py-2">
              <button
                className="layers-panel-section-header flex-1 flex items-center gap-2 text-sm font-medium text-gray-700 dark:text-gray-300 hover:bg-gray-50 dark:hover:bg-gray-700 rounded"
                onClick={() => toggleSection(workflowId)}
              >
                {(expandedSections as any)[workflowId] === false ? <ChevronRight size={14} /> : <ChevronDown size={14} />}
                {editingWorkflow === workflowId ? (
                  <input
                    type="text"
                    value={editingName}
                    onChange={(e) => setEditingName(e.target.value)}
                    onBlur={() => handleSaveRename(workflowId)}
                    onKeyDown={(e) => {
                      if (e.key === 'Enter') {
                        handleSaveRename(workflowId);
                      } else if (e.key === 'Escape') {
                        handleCancelRename();

                    className="bg-transparent border border-gray-300 dark:border-gray-600 rounded px-1 outline-none text-sm"
                    onClick={(e) => e.stopPropagation()}
                    autoFocus
                  />
                ) : (
                  <span>{workflow.name} ({workflow.nodes.length + workflow.edges.length})</span>
                )}

      if (node && node.position) {
        // Focus camera on selected node

          y: -node.position.y + window.innerHeight / 2,
          zoom: Math.max(viewport.zoom, 1.2) // Zoom in if needed
        });
        
        // Select the node
        setSelectedNodes([id]);
        setSelectedEdges([]);
      }
    } else if (type === 'edge') {
      const edge = edges.find(e => e.id === id);
      if (edge) {
        // Find source and target nodes to center camera
        const sourceNode = nodes.find(n => n.id === edge.source);
        const targetNode = nodes.find(n => n.id === edge.target);
        
        if (sourceNode && targetNode) {
          const centerX = (sourceNode.position.x + targetNode.position.x) / 2;
          const centerY = (sourceNode.position.y + targetNode.position.y) / 2;
          
          setViewport({
            x: -centerX + window.innerWidth / 2,
            y: -centerY + window.innerHeight / 2,
            zoom: Math.max(viewport.zoom, 1.0)
          });
          
          // Select the edge
          setSelectedEdges([id]);
          setSelectedNodes([]);
        }
      }
    } else if (type === 'text') {
      const textObj = canvasTexts.find(t => t.id === id);
      if (textObj && textObj.position) {
        setViewport({
          x: -textObj.position.x + window.innerWidth / 2,
          y: -textObj.position.y + window.innerHeight / 2,
          zoom: Math.max(viewport.zoom, 1.2)
        });
        
        // Select the text object
        setSelectedCanvasTexts([id]);
      }
    } else if (type === 'kframe') {
      const kframe = nodes.find(n => n.id === id && n.type === 'kframe');
      if (kframe && kframe.position) {
        setViewport({
          x: -kframe.position.x + window.innerWidth / 2,
          y: -kframe.position.y + window.innerHeight / 2,
          zoom: Math.min(viewport.zoom, 0.8) // Zoom out to see full KFrame
        });
        
        setSelectedNodes([id]);
        setSelectedEdges([]);
      }
    }
    
    onLayerSelect?.(id, type);
  };

  return (
    <div className="kiteframe-canvas-container flex">
      {/* Main Canvas */}
      <div className="flex-1 relative">

            onSelect={handleLayerSelect}
            selectedItemId={selectedLayerItem}
            onRenameWorkflow={(id, name) => {
              console.log('Rename workflow:', id, 'to:', name);
              // Implement workflow renaming logic
            }}
            onToggleWorkflowVisibility={(id) => {
              console.log('Toggle workflow visibility:', id);
              // Implement workflow visibility toggle
            }}
            onDeleteWorkflow={(id) => {
              console.log('Delete workflow:', id);
              // Implement workflow deletion with confirmation
              const workflowNodes = layerStructure.find(w => w.id === id)?.children
                .filter(c => c.type === 'node')
                .map(c => c.id) || [];
              
              // Remove all nodes in the workflow
              setNodes(prev => prev.filter(n => !workflowNodes.includes(n.id)));
              
              // Remove all edges connecting to these nodes
              setEdges(prev => prev.filter(e => 
                !workflowNodes.includes(e.source) && !workflowNodes.includes(e.target)
              ));
            }}
          />
        </div>
      )}
    </div>
  );
}
