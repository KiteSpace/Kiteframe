3) PARSE FIGMA URL
===========================================================

File: client/src/lib/integration/figmaUrl.ts (NEW)

Implement:

export interface ParsedFigmaUrl {
  fileKey: string;
  nodeId?: string;
  rawUrl: string;
}

export function parseFigmaUrl(url: string): ParsedFigmaUrl | null {
  try {
    const u = new URL(url.trim());
    // Figma patterns: /file/<fileKey>/... or /design/<fileKey>/...
    const segments = u.pathname.split('/').filter(Boolean);
    const fileIndex = segments.findIndex(seg => seg === 'file' || seg === 'design');
    if (fileIndex === -1 || !segments[fileIndex + 1]) return null;

    const fileKey = segments[fileIndex + 1];
    const nodeId = u.searchParams.get('node-id') ?? undefined;

    return { fileKey, nodeId, rawUrl: url.trim() };
  } catch {
    return null;
  }
}

import { Node, Edge } from '@/lib/kiteframe/types';
import { ParsedFigmaUrl } from '@/lib/integration/figmaUrl';

export interface WorkflowData {
  nodes: Node[];
  edges: Edge[];
  canvasObjects?: any[];
  viewport?: { x: number; y: number; zoom: number };
}

export function buildFigmaWebviewWorkflow(parsed: ParsedFigmaUrl): WorkflowData {
  const id = `figma-${parsed.fileKey}-${parsed.nodeId ?? 'root'}`;

  const node: Node = {
    id,
    type: 'webview',
    position: { x: 0, y: 0 },
    data: {
      src: parsed.rawUrl,
      title: 'Figma Import',
      kind: 'figma-frame',
      fileKey: parsed.fileKey,
      nodeId: parsed.nodeId,
    },
    draggable: true,
    selectable: true,
    showHandles: false,
  };

  return {
    nodes: [node],
    edges: [],
    canvasObjects: [],
    viewport: { x: 0, y: 0, zoom: 1 },
  };
}
