3. ADD STATUS CYCLING UTILITY
------------------------------------------------------------

Create a utility file:

export function cycleStatus(current?: string): 'todo' | 'inprogress' | 'done' {
  if (!current || current === 'todo') return 'inprogress';
  if (current === 'inprogress') return 'done';
  return 'todo';
}

4. CANVAS STATE UPDATE HANDLER
------------------------------------------------------------
Where node updates are handled (usually in Editor or CanvasController):

Add:

function handleNodeStatusClick(nodeId: string) {
  setNodes(prev =>
    prev.map(n =>
      n.id === nodeId
        ? { ...n, data: { ...n.data, status: cycleStatus(n.data?.status) }}
        : n
    )
  );
}
