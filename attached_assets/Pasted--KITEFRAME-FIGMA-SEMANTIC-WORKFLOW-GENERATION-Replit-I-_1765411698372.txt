SECTION 4 — GENERATION RULES FOR WORKFLOW NODES

The workflow generator MUST follow deterministic rules:

Create a parent workflow group:

workflowGroupId = wf-${frameId}

name = ${frameName} Workflow

For each semantic element:

headings → BasicNode with bold text

text → BasicNode with normal text

labels → BasicNode labeled accordingly

buttons → BasicNode with button-style appearance

inputs → FormNode (if available) or BasicNode with input styling

navigationTargets → create edges connecting logical screens (Phase 1: not implemented)

Node placement strategy:

Vertical stack layout

24px spacing

Starting position = original ImageNode.x + 400 (to avoid overlap)

Required fields to populate in the node:
node: {
id, type, position, data: { label, semanticType, figmaSourceId }
}

Edges:
For MVP, skip actual navigation edges.
Generate an empty edges array.

FILE: client/src/lib/integration/semanticWorkflowGenerator.ts

import { Node, Edge } from "../kiteframe/types";
import type { FigmaSemanticMetadata, FigmaSemanticElement } from "./figmaSemanticTypes";
import { generateId } from "../kiteframe/id";

export function generateWorkflowFromFigmaSemantic(
semantic: FigmaSemanticMetadata,
frameName: string,
sourceNode: Node
): { nodes: Node[]; edges: Edge[]; workflowGroupId: string } {

const workflowGroupId = wf-${sourceNode.data.figmaId || generateId()};

const nodes: Node[] = [];
const edges: Edge[] = [];

// Layout strategy
let x = sourceNode.position.x + (sourceNode.style?.width || 400) + 120;
let y = sourceNode.position.y;

for (const element of semantic.elements || []) {
const node = createNodeFromSemanticElement(element, { x, y }, workflowGroupId);
nodes.push(node);
y += 120; // vertical spacing
}

return { nodes, edges, workflowGroupId };
}

// Helper to create specific node types
function createNodeFromSemanticElement(
element: FigmaSemanticElement,
position: { x: number; y: number },
workflowGroupId: string
): Node {

const baseData = {
label: element.name || element.type,
semanticType: element.type,
figmaElementId: element.id,
workflowGroupId
};

const common = {
id: generateId(),
position,
draggable: true,
selectable: true,
showHandles: true,
};

switch (element.type) {
case "heading":
return {
...common,
type: "basic",
data: { ...baseData, fontWeight: 700, fontSize: 18 }
};

case "text":
case "label":
  return {
    ...common,
    type: "basic",
    data: { ...baseData }
  };

case "button":
  return {
    ...common,
    type: "basic",
    data: { ...baseData, isButton: true }
  };

case "input":
  return {
    ...common,
    type: "form",
    data: { ...baseData, placeholder: element.name || "Input" }
  };

default:
  return {
    ...common,
    type: "basic",
    data: { ...baseData }
  };


}
}
