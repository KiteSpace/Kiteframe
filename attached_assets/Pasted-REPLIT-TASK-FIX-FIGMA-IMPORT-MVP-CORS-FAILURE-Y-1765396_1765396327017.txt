export function parseFigmaUrl(url: string) {
  const keyMatch = url.match(/figma\.com\/file\/([a-zA-Z0-9]+)\//);
  const fileKey = keyMatch?.[1] ?? null;

  const nodeId = (() => {
    try {
      return new URL(url).searchParams.get("node-id");
    } catch {
      return null;
    }
  })();

  return { fileKey, nodeId };
}

STEP 3 — IMPLEMENT FIGMA API FETCHER (NO IFRAME)

import { FIGMA_TOKEN } from "@/config";

export async function fetchFigmaFile(fileKey: string) {
  const res = await fetch(`https://api.figma.com/v1/files/${fileKey}`, {
    headers: { "X-Figma-Token": FIGMA_TOKEN }
  });
  if (!res.ok) throw new Error("Failed to fetch Figma file");
  return res.json();
}

export async function fetchFigmaNode(fileKey: string, nodeId: string) {
  const res = await fetch(
    `https://api.figma.com/v1/files/${fileKey}/nodes?ids=${nodeId}`,
    { headers: { "X-Figma-Token": FIGMA_TOKEN } }
  );
  if (!res.ok) throw new Error("Failed to fetch Figma node");
  return res.json();
}

export async function fetchFigmaThumbnail(fileKey: string, ids: string[]) {
  const res = await fetch(
    `https://api.figma.com/v1/images/${fileKey}?ids=${ids.join(",")}&format=png`,
    { headers: { "X-Figma-Token": FIGMA_TOKEN } }
  );
  if (!res.ok) throw new Error("Failed to fetch Figma thumbnails");
  return res.json();
}

STEP 5 — CONVERT SELECTED FRAME TO KITEFRAME NODES
============================================================

Create file: src/utils/figma/convertFrameToKiteNodes.ts

Rules:
- The selected Figma FRAME becomes one Compound Node.
- Its children become BasicNode or ImageNode nodes.
- Store positions relative to frame coordinates.

Minimal implementation:

export function convertFrameToKiteNodes(frame) {
  const nodes = [];
  const edges = [];

  const compoundId = "figma-frame-" + frame.id;

  nodes.push({
    id: compoundId,
    type: "compound",
    position: { x: 0, y: 0 },
    data: { title: frame.name }
  });

  for (const child of frame.children ?? []) {
    if (child.type === "TEXT") {
      nodes.push({
        id: child.id,
        type: "basic",
        position: { x: child.absoluteBoundingBox.x, y: child.absoluteBoundingBox.y },
        data: { text: child.characters }
      });
    }

    if (child.type === "RECTANGLE") {
      nodes.push({
        id: child.id,
        type: "basic",
        position: { x: child.absoluteBoundingBox.x, y: child.absoluteBoundingBox.y },
        data: { shape: "rect" }
      });
    }

    if (child.type === "IMAGE") {
      nodes.push({
        id: child.id,
        type: "image",
        position: { x: child.absoluteBoundingBox.x, y: child.absoluteBoundingBox.y },
        data: { url: /* retrieve from paints */ "" }
      });
    }
  }

  return { nodes, edges };
}
