SECTION 2 — CENTRALIZE FAST PATH DECISION
--------------------------------------------------

FILE:
client/src/ai/kiteaiState.ts

TASK:
Create a single authoritative fast-path gate.

ADD FUNCTION:

export function shouldExecuteFastPath(
  context: ConversationContext,
  trigger: 'button' | 'phrase' | 'experiment' | 'image'
): boolean {
  if (trigger === 'button') return true;
  if (trigger === 'experiment') return true;
  if (trigger === 'image') return true;
  if (trigger === 'phrase' && isConfirmationPhrase(context.lastUserMessage)) {
    return true;
  }
  if (context.executionTriggered) return true;
  return false;
}

NOTES:
- This function is the ONLY place allowed to bypass clarification
- Confidence score MUST NOT override this
