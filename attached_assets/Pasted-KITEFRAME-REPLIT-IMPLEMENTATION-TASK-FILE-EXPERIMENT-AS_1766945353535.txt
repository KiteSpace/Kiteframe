--------------------------------------------------

A1. DEFINE A NEW WORKFLOW TOOL DATA MODEL

Create a new type (DO NOT reuse Node or CanvasObject):

File: client/src/lib/kiteframe/types.ts

Add:

export type WorkflowToolType = 'experiment';

export interface WorkflowTool {
  id: string;
  type: 'experiment';

  anchorNodeId: string; // node this tool is attached to

  mode: 'what-if' | 'risk' | 'enhancement' | 'prompt';

  state: 'idle' | 'generating' | 'preview';

  selectedOption?: {
    id: string;
    label: string;
    description?: string;
  };

  userPrompt?: string;

  generated?: {
    nodeIds: string[];
    edgeIds: string[];
  };

  meta: {
    experimentId: string;
    source: 'diagnostic' | 'user';
    createdAt: number;
  };
}

IMPORTANT:
- WorkflowTool is NOT a Node
- WorkflowTool is NOT a CanvasObject
- It must not appear in PRDs, diagnostics, or exports
